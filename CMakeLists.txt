cmake_minimum_required(VERSION 3.20)
project(lingua_cpp_jni LANGUAGES CXX)
find_package(Threads REQUIRED)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
if(MSVC)
  # Published JNI DLLs must not require a separately installed VC runtime.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(lingua src/model.cpp src/detector.cpp)
target_compile_features(lingua PUBLIC cxx_std_20)
target_include_directories(lingua PUBLIC include generated)
set_target_properties(lingua PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(lingua PRIVATE /W4)
else()
  target_compile_options(lingua PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(lingua_tool tools/main.cpp)
target_link_libraries(lingua_tool PRIVATE lingua Threads::Threads)
option(LINGUA_BUILD_EXAMPLES "Build examples" ON)
if(LINGUA_BUILD_EXAMPLES)
  add_executable(lingua_example src/examples/service.cpp)
  target_link_libraries(lingua_example PRIVATE lingua Threads::Threads)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(lingua_scoring_test src/tests/scoring.cpp)
  target_link_libraries(lingua_scoring_test PRIVATE lingua)
  add_test(NAME scoring COMMAND lingua_scoring_test)
  add_test(NAME regression COMMAND lingua_tool selftest ${CMAKE_SOURCE_DIR}/data/model.bin)
endif()

option(LINGUA_BUILD_JNI "Build the high-throughput Java adapter" ON)
if(LINGUA_BUILD_JNI)
  find_package(JNI REQUIRED COMPONENTS JVM)
  add_library(lingua_jni SHARED jni/native.cpp)
  target_compile_features(lingua_jni PRIVATE cxx_std_20)
  target_include_directories(lingua_jni PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(lingua_jni PRIVATE lingua)
  if(MSVC)
    target_compile_options(lingua_jni PRIVATE /W4)
  else()
    target_compile_options(lingua_jni PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_link_options(lingua_jni PRIVATE -static-libgcc -static-libstdc++)
    endif()
  endif()
endif()
